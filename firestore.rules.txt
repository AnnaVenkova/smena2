rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Прогресс пользователей: читать и писать может любой,
    // у кого есть ссылка на приложение (без сложной авторизации).
    match /users/{userId} {
      allow read, write: if true;
    }

    // Контент курсов (уроки, тесты): читать может любой,
    // писать (то есть редактировать курсы) — тоже любой с доступом к приложению.
    // Реальная защита при этом — PIN-код внутри самого приложения (Профиль → Режим редактирования),
    // но он не проверяется на стороне Firebase, а только в интерфейсе.
    match /content/{docId} {
      allow read, write: if true;
    }
  }
}

// ВАЖНО: это открытые правила, подходящие для небольшого внутреннего проекта
// (никто посторонний не знает адрес вашей базы данных и не будет туда стучаться).
// Если приложением будет пользоваться много людей извне или данные станут
// чувствительными — потребуется добавить настоящую Firebase Authentication
// и переписать правила так, чтобы пользователь мог писать только в свой документ,
// а редактировать контент могли только конкретные email-адреса администраторов.
