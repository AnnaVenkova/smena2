rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Прогресс пользователей.
    // Читать список всех пользователей (для админ-панели) может только вошедший администратор.
    // Писать свой прогресс могут все — сотрудники не создают отдельный логин,
    // они просто вводят имя при первом запуске (см. README, раздел про остаточные риски).
    match /users/{userId} {
      allow read: if request.auth != null;
      allow write: if true;
    }

    // Контент курсов и портала (уроки, тесты, чек-листы).
    // Читать может любой — иначе приложение не сможет показать курсы сотрудникам.
    // Редактировать (создавать/менять/удалять) может только вошедший администратор.
    match /content/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}

// Что изменилось по сравнению с открытыми правилами:
// - Редактировать курсы и портал теперь может только тот, кто вошёл через
//   Firebase Authentication (email + пароль) — обычный человек с ссылкой
//   на сайт этого сделать не может, даже зная адрес базы данных.
// - Список прогресса всех пользователей тоже виден только вошедшему администратору.
//
// Остаточный риск: запись прогресса в users/{userId} по-прежнему открыта без
// входа — это осознанный компромисс, чтобы сотрудники проходили обучение без
// регистрации отдельного аккаунта. В теории кто-то технически подкованный
// может отправить фальшивый прогресс от чужого имени, зная его userId. Для
// внутреннего учебного приложения такой риск обычно приемлем; если нужно
// закрыть и это — потребуется вход (хотя бы по номеру телефона или коду)
// для каждого сотрудника, а не только для администратора.
