rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Прогресс пользователей.
    // Читать список всех пользователей (для админ-панели) может только вошедший администратор.
    // Создавать/обновлять свой прогресс могут все — сотрудники не создают отдельный логин,
    // они просто вводят имя при первом запуске.
    // А вот УДАЛИТЬ профиль пользователя может только вошедший администратор.
    match /users/{userId} {
      allow read: if request.auth != null;
      allow create, update: if true;
      allow delete: if request.auth != null;
    }

    // Контент курсов и портала (уроки, тесты, чек-листы).
    // Читать может любой — иначе приложение не сможет показать курсы сотрудникам.
    // Редактировать (создавать/менять/удалять) может только вошедший администратор.
    match /content/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}

// Остаточный риск: создание/обновление прогресса в users/{userId} по-прежнему
// открыто без входа — иначе пришлось бы заводить логин каждому сотруднику,
// а не только администратору. Удаление профиля защищено входом.
