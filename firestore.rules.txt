rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Прогресс пользователей.
    // Читать список всех пользователей (для админ-панели) может только вошедший администратор.
    // Создавать/обновлять свой прогресс могут все — сотрудники не создают отдельный логин,
    // они просто вводят имя при первом запуске. Но данные проверяются на базовую валидность,
    // чтобы нельзя было записать туда произвольный мусор или огромные объекты.
    // Удалить профиль пользователя может только вошедший администратор.
    match /users/{userId} {
      allow read: if request.auth != null;

      allow create, update: if
        userId.matches('u_.*') &&
        request.resource.data.name is string &&
        request.resource.data.name.size() > 0 &&
        request.resource.data.name.size() < 100 &&
        request.resource.data.xp is number &&
        request.resource.data.xp >= 0 &&
        request.resource.data.xp < 1000000 &&
        request.resource.data.badges is list &&
        request.resource.data.badges.size() < 50 &&
        (request.resource.data.streak is number) &&
        request.resource.data.streak >= 0 &&
        request.resource.data.streak < 100000;

      allow delete: if request.auth != null;
    }

    // Контент курсов и портала (уроки, тесты, чек-листы).
    // Читать может любой — иначе приложение не сможет показать курсы сотрудникам.
    // Редактировать (создавать/менять/удалять) может только вошедший администратор.
    match /content/{docId} {
      allow read: if true;
      allow write: if request.auth != null;
    }
  }
}

// Что проверяют новые правила для users/{userId}:
// - id документа должен начинаться с "u_" (так их создаёт само приложение) —
//   нельзя записать данные в произвольно названный документ;
// - имя — непустая строка разумной длины (не мегабайты мусора);
// - XP, серия дней и список значков — разумного размера и типа.
// Это не даёт защиты от решительного злоумышленника, который специально
// подделает "правильные на вид" данные, но полностью отсекает случайный
// и автоматический спам/порчу данных.
